Function binarySearch(catalog, productName):
    low = 0
    high = catalog.length - 1
    while low <= high:
        mid = (low + high) / 2
        if catalog[mid] == productName → return FOUND
        else if productName < catalog[mid] → high = mid - 1
        else → low = mid + 1
    return NOT FOUND

    catalog = [AirPods, Backpack, Camera, Laptop, Shoes, Smartphone, Watch]

    Ali searches "Shoes" → FOUND
