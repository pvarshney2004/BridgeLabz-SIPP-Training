enum TrafficLight { RED, YELLOW, GREEN }

class TrafficSignal {
    attribute currentLight
    attribute timer

    constructor() {
        currentLight = RED
        timer = 0
    }

    method switchLight() {
        if currentLight == RED
            currentLight = GREEN
        else if currentLight == GREEN
            currentLight = YELLOW
        else if currentLight == YELLOW
            currentLight = RED
    }

    method runSimulation(totalTime) {
        for t = 1 to totalTime
            if t % 5 == 0        // switch every 5 seconds
                switchLight()
            print("Time: " + t + "s, Light: " + currentLight)
    }
}

// Main
signal = new TrafficSignal()
signal.runSimulation(20)
